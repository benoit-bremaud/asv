class Chatter {
  constructor(name, ip) { this.name = name; this.ip = ip; this.seq = 0; this.inbox = []; this.waiters = []; this.closed = null; }
  open() {
    return new Promise((resolve) => {
      this.ws = new WebSocket(URL, { origin: ORIGIN, localAddress: this.ip });
      this.ws.on("open", () => resolve(true));
      this.ws.on("unexpected-response", (_q, res) => { this.httpStatus = res.statusCode; this.retryAfter = res.headers["retry-after"]; resolve(false); });
      this.ws.on("error", () => resolve(false));
      this.ws.on("message", (d) => {
        const t = d.toString(); if (t === "pong") return;
        const m = JSON.parse(t); this.inbox.push(m);
        this.waiters = this.waiters.filter((w) => (w.pred(m) ? (w.res(m), false) : true));
      });
      this.ws.on("close", (code, reason) => { this.closed = { code, reason: reason.toString() }; });
    });
  }
  send(type, payload) { const seq = ++this.seq; this.ws.send(JSON.stringify({ type, seq, payload })); return seq; }
  raw(t) { this.ws.send(t); }
  wait(pred, ms = 3000) {
    const hit = this.inbox.find(pred); if (hit) { this.inbox.splice(this.inbox.indexOf(hit), 1); return Promise.resolve(hit); }
    return new Promise((res) => { const w = { pred, res: (m) => { this.inbox.splice(this.inbox.indexOf(m), 1); res(m); } }; this.waiters.push(w); setTimeout(() => { this.waiters = this.waiters.filter((x) => x !== w); res(null); }, ms); });
  }
  waitType(type, ms) { return this.wait((m) => m.type === type, ms); }
  drain() { this.inbox = []; }
  count(pred) { return this.inbox.filter(pred).length; }
}
